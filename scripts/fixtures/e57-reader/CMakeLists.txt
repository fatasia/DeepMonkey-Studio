cmake_minimum_required(VERSION 3.15)
project(IndustrialE57ReaderQualification LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
if(NOT E57_SOURCE OR NOT E57_BUILD OR NOT XERCES_INSTALL)
  message(FATAL_ERROR "Set E57_SOURCE, E57_BUILD and XERCES_INSTALL to fixed local builds")
endif()
add_executable(e57-reader main.cpp)
add_executable(e57-attribute-fixture attribute_fixture.cpp)
foreach(target e57-reader e57-attribute-fixture)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE E57_DLL= XERCES_STATIC_LIBRARY)
  target_compile_options(${target} PRIVATE /EHsc /utf-8 /W4)
  target_include_directories(${target} PRIVATE "${E57_SOURCE}/include" "${E57_BUILD}")
  target_link_libraries(${target} PRIVATE "${E57_BUILD}/Release/E57Format.lib" "${XERCES_INSTALL}/lib/xerces-c_3.lib" ws2_32)
endforeach()
